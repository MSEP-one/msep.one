## COMPLETE DNA STRUCTURE BUILDER
## This replaces your current DnaBuilder with full backbone support

func build_complete_dna_structure(
			in_sequence: String,
			in_params := DnaBuilder.Parameters.new()
		) -> Dictionary:
	
	var structure := AtomicStructure.create()
	var atom_tracker := AtomTracker.new()  # Tracks atoms for bonding
	
	structure.start_edit()
	
	var strand_count: int = 2 if in_params.double_strands else 1
	
	for strand in strand_count:
		var sequence = in_sequence if strand == 0 else _get_complement_sequence(in_sequence)
		
		# Build each nucleotide in the strand
		for i in sequence.length():
			var base_type: String = sequence[i]
			var is_5prime_terminal: bool = (i == 0)
			var is_3prime_terminal: bool = (i == sequence.length() - 1)
			
			# Build complete nucleotide (base + sugar + phosphate)
			var nucleotide = _build_complete_nucleotide(
				base_type, i, in_params, strand, is_5prime_terminal
			)
			
			# Add all atoms to structure
			var atom_ids = _add_nucleotide_to_structure(
				structure, nucleotide, base_type, i, strand
			)
			
			# Track atoms for inter-nucleotide bonding
			atom_tracker.add_nucleotide(strand, i, atom_ids)
			
			# Connect to previous nucleotide in strand (phosphodiester bond)
			if i > 0:
				_connect_to_previous_nucleotide(
					structure, atom_tracker, strand, i
				)
	
	# Add hydrogen bonds between complementary bases
	if in_params.double_strands:
		_add_hydrogen_bonds(structure, atom_tracker, in_sequence.length())
	
	structure.end_edit()
	
	return {
		"structure": structure,
		"atom_tracker": atom_tracker
	}


## Add all atoms from a nucleotide to the structure
func _add_nucleotide_to_structure(
			structure: AtomicStructure,
			nucleotide: NucleotideComponents,
			base_type: String,
			position: int,
			strand: int
		) -> Dictionary:
	
	var atom_ids := {
		"base": {},
		"sugar": {},
		"phosphate": {}
	}
	
	var base_start: int = structure.get_valid_atoms_count()
	
	# Add base atoms
	for i in nucleotide.base_atoms.size():
		var atom_data: Vector4 = nucleotide.base_atoms[i]
		var atom_id = structure.add_atom(
			AtomicStructure.AddAtomParameters.new(
				int(atom_data.w),
				Vector3(atom_data.x, atom_data.y, atom_data.z)
			)
		)
		var atom_name = _get_base_atom_name(base_type, i)
		atom_ids.base[atom_name] = atom_id
	
	# Add base bonds
	for bond in nucleotide.base_bonds:
		structure.add_bond(
			base_start + bond.x,
			base_start + bond.y,
			bond.z
		)
	
	var sugar_start: int = structure.get_valid_atoms_count()
	
	# Add sugar atoms
	for i in nucleotide.sugar_atoms.size():
		var atom_data: Vector4 = nucleotide.sugar_atoms[i]
		var atom_id = structure.add_atom(
			AtomicStructure.AddAtomParameters.new(
				int(atom_data.w),
				Vector3(atom_data.x, atom_data.y, atom_data.z)
			)
		)
		var atom_name = _get_sugar_atom_name(i)
		atom_ids.sugar[atom_name] = atom_id
	
	# Add sugar bonds
	for bond in nucleotide.sugar_bonds:
		structure.add_bond(
			sugar_start + bond.x,
			sugar_start + bond.y,
			bond.z
		)
	
	# Connect base to sugar (glycosidic bond)
	var base_connection = "N9" if base_type in ["A", "G"] else "N1"
	var sugar_connection = "C1'"
	
	if base_connection in atom_ids.base and sugar_connection in atom_ids.sugar:
		structure.add_bond(
			atom_ids.base[base_connection],
			atom_ids.sugar[sugar_connection],
			BOND_ORDERS.glycosidic
		)
	
	# Add phosphate (if present)
	if nucleotide.has_5prime_phosphate:
		var phosphate_start: int = structure.get_valid_atoms_count()
		
		for i in nucleotide.phosphate_atoms.size():
			var atom_data: Vector4 = nucleotide.phosphate_atoms[i]
			var atom_id = structure.add_atom(
				AtomicStructure.AddAtomParameters.new(
					int(atom_data.w),
					Vector3(atom_data.x, atom_data.y, atom_data.z)
				)
			)
			var atom_name = _get_phosphate_atom_name(i)
			atom_ids.phosphate[atom_name] = atom_id
		
		# Add phosphate bonds
		for bond in nucleotide.phosphate_bonds:
			structure.add_bond(
				phosphate_start + bond.x,
				phosphate_start + bond.y,
				bond.z
			)
		
		# Connect phosphate to sugar 5' (phosphoester bond)
		if "P" in atom_ids.phosphate and "C5'" in atom_ids.sugar:
			structure.add_bond(
				atom_ids.phosphate["P"],
				atom_ids.sugar["C5'"],
				BOND_ORDERS.phosphoester_5
			)
	
	return atom_ids


## Connect current nucleotide to previous one via phosphodiester bond
func _connect_to_previous_nucleotide(
			structure: AtomicStructure,
			tracker: AtomTracker,
			strand: int,
			current_position: int
		) -> void:
	
	var prev_position = current_position - 1
	
	var prev_atoms = tracker.get_nucleotide_atoms(strand, prev_position)
	var curr_atoms = tracker.get_nucleotide_atoms(strand, current_position)
	
	# Connect: previous sugar C3' -> current phosphate P -> current sugar C5'
	# The backbone: ...Sugar(3')-O-P-O-Sugar(5')...
	
	if "C3'" in prev_atoms.sugar and "P" in curr_atoms.phosphate:
		structure.add_bond(
			prev_atoms.sugar["C3'"],
			curr_atoms.phosphate["P"],
			BOND_ORDERS.phosphoester_3
		)


## Add hydrogen bonds between complementary base pairs
func _add_hydrogen_bonds(
			structure: AtomicStructure,
			tracker: AtomTracker,
			bases_per_strand: int
		) -> void:
	
	for position in bases_per_strand:
		var strand0_atoms = tracker.get_nucleotide_atoms(0, position)
		var strand1_atoms = tracker.get_nucleotide_atoms(1, position)
		
		# Get base types
		var base0_type = tracker.get_base_type(0, position)
		var base1_type = tracker.get_base_type(1, position)
		
		# Add hydrogen bonds based on Watson-Crick pairing
		_add_watson_crick_bonds(
			structure,
			strand0_atoms.base,
			strand1_atoms.base,
			base0_type,
			base1_type
		)


## Add Watson-Crick hydrogen bonds between specific base pair
func _add_watson_crick_bonds(
			structure: AtomicStructure,
			base1_atoms: Dictionary,
			base2_atoms: Dictionary,
			base1_type: String,
			base2_type: String
		) -> void:
	
	# A-T pairing (2 hydrogen bonds)
	if (base1_type == "A" and base2_type == "T") or \
	   (base1_type == "T" and base2_type == "A"):
		if base1_type == "A":
			_add_h_bond_if_exists(structure, base1_atoms, "N6", base2_atoms, "O4")
			_add_h_bond_if_exists(structure, base1_atoms, "N1", base2_atoms, "N3")
		else:
			_add_h_bond_if_exists(structure, base1_atoms, "N3", base2_atoms, "N1")
			_add_h_bond_if_exists(structure, base1_atoms, "O4", base2_atoms, "N6")
	
	# G-C pairing (3 hydrogen bonds)
	elif (base1_type == "G" and base2_type == "C") or \
		 (base1_type == "C" and base2_type == "G"):
		if base1_type == "G":
			_add_h_bond_if_exists(structure, base1_atoms, "N1", base2_atoms, "N3")
			_add_h_bond_if_exists(structure, base1_atoms, "N2", base2_atoms, "O2")
			_add_h_bond_if_exists(structure, base1_atoms, "O6", base2_atoms, "N4")
		else:
			_add_h_bond_if_exists(structure, base1_atoms, "N4", base2_atoms, "O6")
			_add_h_bond_if_exists(structure, base1_atoms, "O2", base2_atoms, "N2")
			_add_h_bond_if_exists(structure, base1_atoms, "N3", base2_atoms, "N1")


func _add_h_bond_if_exists(
			structure: AtomicStructure,
			atoms1: Dictionary,
			name1: String,
			atoms2: Dictionary,
			name2: String
		) -> void:
	if name1 in atoms1 and name2 in atoms2:
		structure.add_bond(
			atoms1[name1],
			atoms2[name2],
			BOND_ORDERS.hydrogen
		)


## Helper class to track atom IDs for bonding
class AtomTracker:
	# Structure: { strand_id: { position: { base: {}, sugar: {}, phosphate: {} } } }
	var nucleotides: Dictionary = {}
	var base_types: Dictionary = {}  # { (strand, position): base_type }
	
	func add_nucleotide(strand: int, position: int, atom_ids: Dictionary) -> void:
		if not strand in nucleotides:
			nucleotides[strand] = {}
		nucleotides[strand][position] = atom_ids
	
	func get_nucleotide_atoms(strand: int, position: int) -> Dictionary:
		return nucleotides[strand][position]
	
	func set_base_type(strand: int, position: int, base_type: String) -> void:
		base_types[[strand, position]] = base_type
	
	func get_base_type(strand: int, position: int) -> String:
		return base_types.get([strand, position], "")


func _get_complement_sequence(sequence: String) -> String:
	const COMPLEMENT = {'A': 'T', 'T': 'A', 'G': 'C', 'C': 'G'}
	var result = ""
	for base in sequence:
		result += COMPLEMENT[base]
	return result

## Simplified: Generate sugar-phosphate backbone programmatically
class SimplifiedBackboneBuilder:
	## Creates idealized sugar and phosphate geometries
	
	const SUGAR_GEOMETRY: Dictionary = {
		# Deoxyribose in chair conformation (approximate coordinates)
		# Relative to base attachment point
		"C1'": Vector3(0.0, 0.0, 0.0),      # Connection to base
		"C2'": Vector3(0.15, -0.13, 0.0),
		"C3'": Vector3(0.08, -0.25, 0.13),  # Connection to phosphate (3' end)
		"C4'": Vector3(-0.12, -0.20, 0.15),
		"O4'": Vector3(-0.15, -0.05, 0.05),
		"C5'": Vector3(-0.23, -0.25, 0.28), # Connection to phosphate (5' end)
	}
	
	const PHOSPHATE_GEOMETRY: Dictionary = {
		# Phosphate group (PO4) geometry
		"P": Vector3(0.0, 0.0, 0.0),
		"O1P": Vector3(0.15, 0.0, 0.0),   # Non-bridging oxygen
		"O2P": Vector3(-0.15, 0.0, 0.0),  # Non-bridging oxygen
		"O5'": Vector3(0.0, 0.17, 0.0),   # Connects to 5' carbon
		"O3'": Vector3(0.0, -0.17, 0.0),  # Connects to 3' carbon (of prev sugar)
	}
	
	func create_sugar_atoms(base_position: Vector3, helix_angle: float) -> Array:
		var atoms = []
		for atom_name in SUGAR_GEOMETRY:
			var local_pos: Vector3 = SUGAR_GEOMETRY[atom_name]
			var rotated_pos = local_pos.rotated(Vector3.FORWARD, helix_angle)
			var final_pos = base_position + rotated_pos
			atoms.append({
				"name": atom_name,
				"position": final_pos,
				"element": 6 if atom_name.begins_with("C") else 8  # C or O
			})
		return atoms
	
	func create_phosphate_atoms(position: Vector3, helix_angle: float) -> Array:
		var atoms = []
		for atom_name in PHOSPHATE_GEOMETRY:
			var local_pos: Vector3 = PHOSPHATE_GEOMETRY[atom_name]
			var rotated_pos = local_pos.rotated(Vector3.FORWARD, helix_angle)
			var final_pos = position + rotated_pos
			var element = 15 if atom_name == "P" else 8  # P or O
			atoms.append({
				"name": atom_name,
				"position": final_pos,
				"element": element
			})
		return atoms


# USAGE SUMMARY:
# ==============
#
# DNA Structure = For each nucleotide:
#   1. Base (A/T/G/C) - aromatic ring structure
#   2. Sugar (deoxyribose) - 5-carbon ring
#   3. Phosphate (PO4) - links to next nucleotide
#
# Bonding:
#   - Base-Sugar: Glycosidic bond (C1'-N9 or C1'-N1)
#   - Sugar-Phosphate: Phosphoester bonds (3' and 5')
#   - Between strands: Hydrogen bonds (A-T: 2 bonds, G-C: 3 bonds)
#
# The backbone: 5'-Sugar-Phosphate-Sugar-Phosphate-Sugar-3'
#                    |         |         |
#                   Base     Base     Base
#                    :         :         :  (hydrogen bonds)
#                   Base     Base     Base
#                    |         |         |
#                3'-Sugar-Phosphate-Sugar-Phosphate-Sugar-5'
